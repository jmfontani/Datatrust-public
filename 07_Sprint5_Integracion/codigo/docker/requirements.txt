# ANTES: el Dockerfile instalaba 13 paquetes SIN NINGUNA VERSION FIJADA:
#
#     RUN pip install --no-cache-dir fastapi uvicorn[standard] psycopg2-binary \
#         pgvector python-dotenv sentence-transformers numpy pyyaml requests \
#         scikit-learn scipy ollama
#
# Consecuencia: cada `docker compose build` en una fecha distinta instala lo
# que haya ese dia. Dos builds con el mismo codigo dan entornos diferentes, y
# si cambia algo del stack de embeddings, los vectores que produzca el
# contenedor dejan de coincidir con los guardados en la base de datos.
#
# Este fichero fija las versiones EXACTAS del entorno que esta actualmente en
# funcionamiento (verificado el 24-sep-2026 contra el contenedor `buscador_api`).
#
# ── COMO APLICARLO ──────────────────────────────────────────────────────────
#
# En Dockerfile.api, sustituir el bloque de 13 paquetes por:
#
#     WORKDIR /app
#     COPY 07_Sprint5_Integracion/codigo/docker/requirements.txt ./requirements.txt
#     RUN pip install --no-cache-dir -r requirements.txt
#
# ── INDICE DE PAQUETES ──────────────────────────────────────────────────────
#
# torch 2.14.0 se instala desde el indice de PyTorch, NO desde PyPI. El wheel
# instalado tiene tag cp311-manylinux_2_28_x86_64 y arrastra las librerias
# nvidia-*-cu13 (CUDA 13). Si se instala desde PyPI puede resolver a una build
# distinta, y eso cambia el resultado del embedding.
#
#   pip install --index-url https://download.pytorch.org/whl/cu130 -r requirements.txt
#
# NOTA IMPORTANTE: este contenedor NO usa GPU (torch.cuda.is_available() es
# False en `buscador_api`). Las dependencias CUDA suman ~5,3 GB de los 10,1 GB
# de la imagen y no aportan nada al calculo de embeddings, que es en CPU. Se
# mantiene esta build concreta porque es la que genero / valida el entorno
# actual; cambiarla a una build CPU es una optimizacion aparte que exige
# volver a validar los embeddings (ver PROPUESTA-BUILD-CPU.md).
